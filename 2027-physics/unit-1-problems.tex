\begin{solution}

\DSTtriangle[shade=D]

\question Find the slope of the line.

\xygraphI{2}{1}{(0,1)(3,7)}


\begin{solution}
Using points $(0,1)$ and $(3,7)$,

\begin{equation*}
    m = \frac{\Delta y}{\Delta x} = \frac{y_2 - y_1}{x_2 - x_1} = \frac{7-1}{3-0} = \boxed{2}
\end{equation*}
\end{solution}
