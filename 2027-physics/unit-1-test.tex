            };
    \end{pgfonlayer}

\begin{randomizechoices}
    \correctchoice $+\SI{9}{m}$
    \choice $-\SI{12}{m}$
    \choice $-\SI{1}{m}$
    \choice $+\SI{33}{m}$
\end{randomizechoices}

\begin{solution}
The equation $\Delta x = \bar{v}t$ implies that displacement is the area bounded by the line.

The first bounded area $A_1$ is

\begin{equation*}
    \Delta x_1 = (\SI{-4}{m/s})(\SI{3}{s}) = \SI{-12}{m}
\end{equation*}

Note that the ball moves in the negative direction in the first 3 seconds.

The second displacement is

\begin{equation*}
    \Delta x_2 = (+\SI{3}{m/s})(\SI{7}{s}) = \SI{21}{m}
\end{equation*}

The total displacement is the sum of these ``areas,'' or

\begin{equation*}
    \Delta x_\mathrm{tot} = \Delta x_1 + \Delta x_2 = \SI{-12}{m} + \SI{21}{m} = \boxed{\SI{9}{m}}
\end{equation*}
\end{solution}
